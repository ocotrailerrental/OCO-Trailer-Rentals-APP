# OCO Trailer Rentals — robots.txt
# Served as a real static file so crawlers get this, not the app shell.

# The Disallow rules have to be repeated in every named-agent block. A crawler
# obeys the FIRST block matching its own name and ignores all the others, so the
# AI blocks below used to carry a bare "Allow: /" and were being invited into the
# very account pages this file exists to keep out of search results.

User-agent: *
Allow: /
# Customer account pages and auth flows hold nothing useful to a crawler and
# should never appear in search results.
Disallow: /app
Disallow: /login
Disallow: /signup
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /reserve

# AI answer engines — allowed on the public pages so OCO can be cited when
# someone asks about trailer rentals in Omaha or Anchorage, and held to the same
# limits as everybody else.

User-agent: OAI-SearchBot
Allow: /
Disallow: /app
Disallow: /login
Disallow: /signup
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /reserve

Allow: /
Disallow: /app
Disallow: /login
Disallow: /signup
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /reserve

User-agent: PerplexityBot
Allow: /
Disallow: /app
Disallow: /login
Disallow: /signup
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /reserve
